        function initMobileTouchListeners() {
            const moveBase = document.getElementById('moveJoystick');
            const moveKnob = document.getElementById('moveJoystickKnob');
            const aimBase = document.getElementById('aimJoystick');
            const aimKnob = document.getElementById('aimJoystickKnob');
            const fireBtn = document.getElementById('mobileFireBtn');

            let moveTouchId = null;
            let aimTouchId = null;
            const maxRadius = 38;

            if (moveBase && moveKnob) {
                moveBase.addEventListener('touchstart', e => {
                    e.preventDefault();
                    sound.init();
                    const t = e.changedTouches[0];
                    moveTouchId = t.identifier;
                    updateMoveStick(t.clientX, t.clientY);
                }, { passive: false });

                moveBase.addEventListener('touchmove', e => {
                    e.preventDefault();
                    for (let i = 0; i < e.changedTouches.length; i++) {
                        const t = e.changedTouches[i];
                        if (t.identifier === moveTouchId) {
                            updateMoveStick(t.clientX, t.clientY);
                        }
                    }
                }, { passive: false });

                const resetMove = e => {
                    for (let i = 0; i < e.changedTouches.length; i++) {
                        if (e.changedTouches[i].identifier === moveTouchId) {
                            moveTouchId = null;
                            touchKeys.joyX = 0;
                            touchKeys.joyY = 0;
                            moveKnob.style.transform = 'translate(-50%, -50%)';
                        }
                    }
                };
                moveBase.addEventListener('touchend', resetMove);
                moveBase.addEventListener('touchcancel', resetMove);

                function updateMoveStick(clientX, clientY) {
                    const rect = moveBase.getBoundingClientRect();
                    const cx = rect.left + rect.width / 2;
                    const cy = rect.top + rect.height / 2;
                    let dx = clientX - cx;
                    let dy = clientY - cy;
                    const dist = Math.hypot(dx, dy);
                    if (dist > maxRadius) {
                        dx = (dx / dist) * maxRadius;
                        dy = (dy / dist) * maxRadius;
                    }
                    moveKnob.style.transform = `translate(calc(-50% + ${dx}px), calc(-50% + ${dy}px))`;
                    touchKeys.joyX = dx / maxRadius;
                    touchKeys.joyY = dy / maxRadius;
                }
            }

            if (aimBase && aimKnob) {
                aimBase.addEventListener('touchstart', e => {
                    e.preventDefault();
                    sound.init();
                    const t = e.changedTouches[0];
                    aimTouchId = t.identifier;
                    touchKeys.aimActive = true;
                    updateAimStick(t.clientX, t.clientY);
                }, { passive: false });

                aimBase.addEventListener('touchmove', e => {
                    e.preventDefault();

                        }
                    }
                }, { passive: false });

                const resetAim = e => {

                            aimTouchId = null;
                            touchKeys.aimActive = false;

                        }
                    }
